const jwt = require('jsonwebtoken');
const authService = require('../services/authService');
const { validateFields } = require('../utils/validation');

// Authenticates user credentials and issues a JWT
exports.login = (req, res, next) => {
  try {
    validateFields(req.body, ['email', 'password']);

    const { email, password } = req.body;

    const userExists = authService.getUserByEmail(email);
    if (!userExists) {
      const error = new Error("User not found");
      error.status = 404;
      throw error;
    }

    const user = authService.verifyCredentials(email, password);
    if (!user) {
      const error = new Error("Incorrect password");
      error.status = 401;
      throw error;
    }

    const secret = process.env.JWT_SECRET;
    if (!secret) {
      const error = new Error("JWT_SECRET is not configured");
      error.status = 500;
      throw error;
    }

    const token = jwt.sign(
      { userId: user.id },
      secret,
      { expiresIn: '1h' }
    );

    return res.status(200).json({ token });
  } catch (err) {
    next(err);
  }
};

const jwt = require('jsonwebtoken');
const userRepository = require('../repositories/userRepository');

// Validate JWT from Authorization header and attach the user to req.user.
const requireAuth = (req, res, next) => {
  try {
    const authHeader = req.headers['authorization'];

    if (!authHeader || !authHeader.startsWith('Bearer ')) {
      const error = new Error("Authentication required");
      error.status = 401;
      throw error;
    }

    const token = authHeader.substring('Bearer '.length).trim();
    if (!token) {
      const error = new Error("Authentication required");
      error.status = 401;
      throw error;
    }

    const secret = process.env.JWT_SECRET;
    if (!secret) {
      const error = new Error("JWT_SECRET is not configured");
      error.status = 500;
      throw error;
    }

    const payload = jwt.verify(token, secret);

    const userId = payload?.userId;
    if (!userId) {
      const error = new Error("Invalid token");
      error.status = 401;
      throw error;
    }

    const user = userRepository.findById(userId);
    if (!user) {
      const error = new Error("Invalid token user");
      error.status = 401;
      throw error;
    }

    req.user = user;
    next();
  } catch (err) {
    if (err.name === 'JsonWebTokenError' || err.name === 'TokenExpiredError') {
      err.status = 401;
      err.message = "Invalid or expired token";
    }
    next(err);
  }
};

module.exports = { requireAuth };

const { files } = require('../utils/storage');

const normalizeParentId = (parentId) => (parentId === undefined ? null : parentId);

// Finds non-trashed files within a specific folder
const findByParent = (parentId, ownerId) => {
  const p = normalizeParentId(parentId);
  return files.filter(f => {
    if (f.trashed === true) return false;
    const fp = normalizeParentId(f.parentId);
    const parentMatch = fp === p;
    const ownerMatch = ownerId ? f.ownerId === ownerId : true; 
    return parentMatch && ownerMatch;
  });
};

// Returns all files in storage
const findAll = () => files;

// Finds a file by ID (including trashed files)
const findById = (id, ownerId) => {
  if (!ownerId) return files.find(f => f.id === id);
  return files.find(f => f.id === id && f.ownerId === ownerId);
};

// Creates a new file entry
const create = (fileData) => {
  const newFileData = {
    mimeType: "",
    trashed: false,
    trashedAt: null,
    originalParentId: null,
    lastAccessedBy: {},
    blockedBy: [],
    ...fileData
  };

  files.push(newFileData);
  return newFileData;
};

// Updates an existing file entry
const update = (id, updates, ownerId) => {
  const idx = files.findIndex(f => {
    if (!ownerId) return f.id === id;
    return f.id === id && f.ownerId === ownerId;
  });

  if (idx === -1) return null;

  files[idx] = { ...files[idx], ...updates };
  return files[idx];
};

// Removes a file entry from storage
const remove = (id, ownerId) => {
  const idx = files.findIndex(f => {
    if (!ownerId) return f.id === id;
    return f.id === id && f.ownerId === ownerId;
  });

  if (idx === -1) return null;
  return files.splice(idx, 1)[0];
};

// Checks if a folder contains any non-trashed files
const hasChildren = (parentId, ownerId) => {
  return files.some(f => {
    if (f.trashed === true) return false;
    const ownerMatch = ownerId ? f.ownerId === ownerId : true;
    return ownerMatch && (f.parentId ?? null) === parentId;
  });
};

// Searches for non-trashed files by name or content
const searchFiles = (userId, query) => {
  const normalizedQuery = query.toLowerCase();

  return files.filter(file =>
    file.ownerId === userId &&
    file.trashed !== true &&
    (
      file.name.toLowerCase().includes(normalizedQuery) ||
      (file.content && file.content.toLowerCase().includes(normalizedQuery))
    )
  );
};

const findTrashed = (ownerId) => {
  return files.filter(f => f.ownerId === ownerId && f.trashed === true);
};

// Finds multiple files by an array of IDs
const findByIds = (idsArray) => files.filter(f => idsArray.includes(f.id));

const contentSizeBytes = (content) => {
  if (content === undefined || content === null) return 0;
  const s = typeof content === 'string' ? content : String(content);

const isDescendant = (fileId, folderId) => {
  let current = fileRepository.findById(fileId);
  while (current && current.parentId) {
    if (current.parentId === folderId) return true;
    current = fileRepository.findById(current.parentId);

};

const wouldCreateCycle = (itemId, newParentId) => {
  let cursor = newParentId ?? null;
  while (cursor !== null) {
    if (cursor === itemId) return true;
    const node = fileRepository.findById(cursor);
    if (!node) return false;
    cursor = node.parentId ?? null;
  }
  return false;
};

const collectSubtreeIds = (rootId) => {
  const stack = [rootId];
  const visited = new Set();
  const ids = [];
  while (stack.length > 0) {
    const currentId = stack.pop();
    if (visited.has(currentId)) continue;
    visited.add(currentId);
    ids.push(currentId);
    const children = fileRepository.findByParent(currentId) || [];
    for (const child of children) stack.push(child.id);
  }
  return ids;
};

const getTrashedAtForUser = (item, userId) => {
  const map = item?.trashedAtBy && typeof item.trashedAtBy === 'object' ? item.trashedAtBy : {};

const calcDaysLeft = (trashedAt) => {
  if (!trashedAt) return null;
  const t = new Date(trashedAt).getTime();
  if (Number.isNaN(t)) return null;
  const expireAt = t + TRASH_TTL_MS;
  const msLeft = expireAt - Date.now();
  return Math.max(0, Math.ceil(msLeft / (24 * 60 * 60 * 1000)));
};

const decorateFlagsForUser = (item, userId) => {
  const starredBy = ensureArray(item.starredBy);
  const importantBy = ensureArray(item.importantBy);
  const trashedBy = ensureArray(item.trashedBy);

  const trashed = trashedBy.includes(userId);

  const lastAccessedBy = (item.lastAccessedBy && typeof item.lastAccessedBy === 'object')
    ? item.lastAccessedBy
    : {};

  return {
    ...item,
    starred: starredBy.includes(userId),
    important: importantBy.includes(userId),
    trashed,
    trashedAt,
    trashDaysLeft: trashed ? calcDaysLeft(trashedAt) : null,
    lastAccessed
  };

  const roots = [...ownedRoots];

    const already = roots.some(r => r.id === item.id);
    const hasAncestorInShared = sharedRoots.some(other => other.id !== item.id && isDescendant(item.id, other.id));
    if (!already && !hasAncestorInShared) roots.push(item);
  }

  for (const root of roots) {
    const ids = collectSubtreeIds(root.id);
    for (const id of ids) {
      if (await checkAccess(userId, id, 'viewer')) {

        if (item) result.set(item.id, item);

    }
  }
  const decorated = decorateListForUser(Array.from(result.values()), userId);
  return includeTrashed ? decorated : decorated.filter(i => !i.trashed);

const updateItem = async (userId, id, patch) => {

  if (!current) throw { status: 404, message: 'File or folder not found' };
  if (!(await checkAccess(userId, id, 'editor'))) throw { status: 403, message: 'Editor access required' };

    try {
      try { await sendTcpCommand(`DELETE ${id}`); } catch (e) { }
      await sendTcpCommand(`POST ${id} ${patch.content}`);
      updates.sizeBytes = contentSizeBytes(patch.content);
    } catch (err) { throw err; }
  }

  if (patch.mimeType !== undefined) updates.mimeType = patch.mimeType;

  if (patch.name !== undefined) {
    const trimmed = patch.name.trim();
    if (trimmed === '') throw { status: 400, message: 'Name must be a non-empty string' };
    updates.name = trimmed;
  }

  if (patch.parentId !== undefined) {
    const newParentId = patch.parentId ?? null;
    if (newParentId === id) throw { status: 400, message: 'parentId cannot be the same as id' };

    if (newParentId !== null) {

      if (current.type === 'folder' && wouldCreateCycle(id, newParentId)) {
        throw { status: 409, message: 'Cannot move a folder into its own descendant' };
      }
    }
    updates.parentId = newParentId;
  }

const deleteForever = async (userId, id) => {
  const item = fileRepository.findById(id);

  if (!item) return { id };

  if (!isTrashedForUser(item, userId)) {
    throw { status: 409, message: 'Item not found in your trash' };
  }

  if (item.ownerId !== userId) {
    const ids = item.type === 'folder' ? collectSubtreeIds(id) : [id];

    for (const dId of ids) {
      const target = fileRepository.findById(dId);
      if (!target) continue;

      const trashedBy = ensureArray(target.trashedBy).filter(x => x !== userId);
      const trashedAtBy = { ...(target.trashedAtBy || {}) };
      delete trashedAtBy[userId];

      const blockedBy = [...new Set([...ensureArray(target.blockedBy), userId])];

      fileRepository.update(dId, { trashedBy, trashedAtBy, blockedBy });

      const myPermission = (permissionRepository.findByFileId(dId) || []).find(p => p.userId === userId);
      if (myPermission) {
        permissionRepository.remove(myPermission.id);
      }
    }
    return { id };
  }

  const ids = item.type === 'folder' ? collectSubtreeIds(id).reverse() : [id];
  for (const dId of ids) {
    const target = fileRepository.findById(dId);
    if (!target) continue;

    if (target.type === 'file') {
      await sendTcpCommand(`DELETE ${dId}`).catch(() => { });
    }
    fileRepository.remove(dId);

    const allPerms = permissionRepository.findByFileId(dId) || [];
    allPerms.forEach(p => permissionRepository.remove(p.id));
  }
  return { id };
};

const sendTcpCommand = (command) => {
    return new Promise((resolve, reject) => {
        const client = new net.Socket();
        console.log(`[TCP Client] Connecting to: ${SERVER_HOST}:${SERVER_PORT}`);

        client.connect(SERVER_PORT, SERVER_HOST, () => {
            client.write(command + '\n');
            client.end();
        });

        let receivedData = '';

        client.on('data', (chunk) => {
            receivedData += chunk.toString();
        });

        client.on('end', () => {
            console.log("[TCP Client] Connection closed by server, parsing data...");

            const lines = receivedData.split('\n');
            if (lines.length === 0) return resolve('');

            const statusLine = lines[0].trim();
            const statusCode = parseInt(statusLine.split(' ')[0], 10);

            if (statusCode >= 200 && statusCode < 300) {
                const separatorIndex = lines.indexOf('');

                if (separatorIndex !== -1 && separatorIndex < lines.length - 1) {
                    const body = lines.slice(separatorIndex + 1).join('\n');
                    resolve(body);
                } else {
                    resolve('');
                }
            } else {
                const error = new Error(statusLine);
                error.status = statusCode;
                reject(error);
            }
        });

        client.on('error', (err) => {
            console.error(`[TCP Client] Connection Error to ${SERVER_HOST}:`, err.message);
            const error = new Error('Storage server unavailable');
            error.status = 503;
            reject(error);
        });
    });
};

module.exports = { sendTcpCommand };

// Validates that required fields exist in the request body
const validateFields = (body, requiredFields) => {
  for (const field of requiredFields) {
    if (!body[field]) {
      const error = new Error(`${field.charAt(0).toUpperCase() + field.slice(1)} is required`);
      error.status = 400;
      throw error;
    }
  }
};

// Validates that a field is a non-empty string, throwing 400 if invalid
const validateNonEmptyString = (body, field, message) => {
  const value = body?.[field];
  if (typeof value !== 'string' || value.trim() === '') {
    const err = new Error(message || `${field} must be a non-empty string`);
    err.status = 400;
    throw err;
  }
};

// Checks if a resource exists, throwing 404 if not found
const validateExistence = (resource, resourceName) => {
  if (!resource) {
    const error = new Error(`${resourceName} not found`);
    error.status = 404;
    throw error;
  }
};

// Validates that the date of birth follows the YYYY-MM-DD format
const validateDateOfBirth = (body, field = 'dateOfBirth') => {
  validateNonEmptyString(body, field, `${field} must be a non-empty string`);

  const value = body[field].trim();
  const dobOk = /^\d{4}-\d{2}-\d{2}$/.test(value);

  if (!dobOk) {
    const error = new Error(`${field} must be in YYYY-MM-DD format`);
    error.status = 400;
    throw error;
  }
};

// Validates that the gender field contains an allowed value
const validateGender = (body, field = 'gender') => {
  validateNonEmptyString(body, field, `${field} must be a non-empty string`);

  const value = body[field].trim();
  const allowed = new Set(['male', 'female', 'other', 'prefer_not_to_say']);

  if (!allowed.has(value)) {
    const error = new Error(`${field} must be one of: male, female, other, prefer_not_to_say`);
    error.status = 400;
    throw error;
  }
};

module.exports = {
  validateFields,
  validateNonEmptyString,
  validateExistence,
  validateDateOfBirth,
  validateGender
};

    if (err instanceof SyntaxError && err.status === 400 && 'body' in err) {
