export async function request(path, options = {}) {
  const method = options.method ?? "GET";

  const headers = {
    "Content-Type": "application/json",
    ...(options.headers ?? {}),
  };

  if (options.token) headers.Authorization = `Bearer ${options.token}`;

  const res = await fetch(`${BASE_URL}${path}`, {
    method,
    headers,
    body: options.body ? JSON.stringify(options.body) : undefined,
  });

  const text = await res.text();
  const data = text ? safeJson(text) : null;

  if (!res.ok) {
    const msg =
      (data && (data.message || data.error)) ||
      `Request failed (${res.status})`;
    throw new Error(msg);
  }

  return data;
}

function safeJson(text) {
  try { return JSON.parse(text); } catch { return null; }
}

import React, { createContext, useContext, useEffect, useMemo, useState } from "react";
import * as SecureStore from "expo-secure-store";
import { useRouter } from "expo-router";
import { authenticateUser } from "../../features/auth/services/authService";
import { getUserProfile } from "../../features/user/services/userService";

const TOKEN_KEY = "token";
const USER_KEY = "user_data";
const AuthContext = createContext(null);

/** ׳₪׳•׳ ׳§׳¦׳™׳™׳× ׳¢׳–׳¨ ׳׳₪׳¢׳ ׳•׳— userId ׳׳×׳•׳ ׳”-JWT ׳׳׳ ׳¡׳₪׳¨׳™׳•׳× ׳—׳™׳¦׳•׳ ׳™׳•׳× */
function getUserIdFromToken(token) {
  try {
    const payload = JSON.parse(atob(token.split(".")[1]));
    return payload.userId;
  } catch (e) { return null; }
}

export function AuthProvider({ children }) {
  const [token, setToken] = useState(null);
  const [user, setUser] = useState(null);
  const [isHydrating, setIsHydrating] = useState(true);
  const router = useRouter();

  useEffect(() => {
    (async () => {
      try {
        const savedToken = await SecureStore.getItemAsync(TOKEN_KEY);
        const savedUser = await SecureStore.getItemAsync(USER_KEY);
        if (savedToken) setToken(savedToken);
        if (savedUser) setUser(JSON.parse(savedUser));
        
        // ׳׳ ׳™׳© ׳˜׳•׳§׳ ׳׳‘׳ ׳׳™׳ ׳׳©׳×׳׳© (׳׳׳©׳ ׳׳—׳¨׳™ ׳¨׳™׳¡׳˜׳¨׳˜), ׳ ׳ ׳¡׳” ׳׳׳©׳•׳ ׳׳•׳×׳•
        if (savedToken && !savedUser) {
          const uid = getUserIdFromToken(savedToken);
          const userData = await getUserProfile(savedToken, uid);
          if (userData) {
            setUser(userData);
            await SecureStore.setItemAsync(USER_KEY, JSON.stringify(userData));
          }
        }
      } catch (e) {
        console.error("Hydration error", e);
      } finally {
        setIsHydrating(false);
      }
    })();
  }, []);

  const login = async (email, password) => {
    try {
      const res = await authenticateUser(email, password);
      if (!res.ok) return { ok: false, error: res.error };

      const newToken = res.token;
      const userId = getUserIdFromToken(newToken);
      
      // ׳©׳׳™׳₪׳× ׳ ׳×׳•׳ ׳™ ׳”׳׳©׳×׳׳© ׳׳™׳“ ׳׳׳—׳¨ ׳§׳‘׳׳× ׳”׳˜׳•׳§׳ (׳‘׳“׳™׳•׳§ ׳›׳׳• ׳‘-Web)
      const userData = await getUserProfile(newToken, userId);

      await SecureStore.setItemAsync(TOKEN_KEY, newToken);
      if (userData) await SecureStore.setItemAsync(USER_KEY, JSON.stringify(userData));

      setToken(newToken);
      setUser(userData);
      
      router.replace("/(app)/home");
      return { ok: true };
    } catch (e) {
      return { ok: false, error: "Login failed" };
    }
  };

  const logout = async () => {
    await SecureStore.deleteItemAsync(TOKEN_KEY);
    await SecureStore.deleteItemAsync(USER_KEY);
    setToken(null);
    setUser(null);
    router.replace("/(auth)/login");
  };

  const value = useMemo(() => ({
    token, user, isAuthenticated: !!token, isHydrating, login, logout
  }), [token, user, isHydrating]);

  return <AuthContext.Provider value={value}>{children}</AuthContext.Provider>;
}

async function request(path, options = {}) {
  const method = options.method || "GET";
  const body = options.body;

  const cleanPath = path.startsWith("/api") ? path.replace("/api", "") : path;
  const url = `${API_BASE_URL}${cleanPath}`;

  const headers = {
    "Content-Type": "application/json",
    ...(options.headers || {}),
  };

  const res = await fetch(url, {
    method,
    headers,
    body: body != null ? JSON.stringify(body) : undefined,
  });

  const text = await res.text();
  let data = null;

  try {
    data = text ? JSON.parse(text) : null;
  } catch {
    data = text;
  }

  return { ok: res.ok, status: res.status, data };
}

export async function authenticateUser(email, password) {
  const { ok, data } = await request("/tokens", {
    method: "POST",
    body: { email, password },
  });

  if (!ok) {
    return {
      ok: false,
      error: data?.error || data?.message || "Invalid credentials",
    };
  }

  const token = data?.token || data?.accessToken || data?.jwt;
  return { ok: true, token };
}

    body: payload,

  if (ok || status === 201) {
    return { success: true, user: data };
  }

  return {
    success: false,
    error: data?.error || data?.message || "Register failed",
  };
}

// mobile-client/src/features/drive/hooks/useMyDrive.js

import { useState, useEffect, useCallback } from "react";
import { useAuth } from "../../../app/providers/AuthContext";
import { getFiles } from "../services/fileService";
import { subscribeDriveRefresh } from "../services/driveEvents";

export function useMyDrive() {
  const { token, logout } = useAuth();

  const [items, setItems] = useState([]);
  const [loading, setLoading] = useState(true);
  const [refreshing, setRefreshing] = useState(false);

  const fetchItems = useCallback(async () => {
    if (!token) return;

    try {
      const data = await getFiles(token);
      setItems(Array.isArray(data) ? data : []);
    } catch (error) {
      if (
        error?.message &&
        (error.message.includes("token") || error.message.includes("401"))
      ) {
        await logout();
      }
    } finally {
      setLoading(false);
      setRefreshing(false);
    }
  }, [token, logout]);

  useEffect(() => {
    fetchItems();
  }, [fetchItems]);

  // ג… listen for global refresh events (upload/create/delete/etc.)
  useEffect(() => {
    const unsubscribe = subscribeDriveRefresh(() => {
      setRefreshing(true);
      fetchItems();
    });
    return unsubscribe;
  }, [fetchItems]);

  const onRefresh = () => {
    setRefreshing(true);
    fetchItems();
  };

  return { items, loading, refreshing, onRefresh };
}

export function getDaysLeft(trashedAt) {
    if (!trashedAt) return "";
    const trashDate = new Date(trashedAt);
    const deleteDate = new Date(trashDate.getTime() + 30 * 24 * 60 * 60 * 1000);
    const now = new Date();
    const diffTime = deleteDate - now;
    const diffDays = Math.ceil(diffTime / (1000 * 60 * 60 * 24));
    return diffDays > 0 ? `${diffDays} days left` : "Deleting soon";
}

export function formatLastAccessed(dateString) {
    if (!dateString) return "Never";
    const date = new Date(dateString);
    const diffMs = new Date() - date;
    const diffMins = Math.floor(diffMs / 60000);

    if (diffMins < 1) return "Just now";
    if (diffMins < 60) return `${diffMins} mins ago`;
    if (diffMs < 86400000) return `${Math.floor(diffMs / 3600000)} hours ago`;
    return date.toLocaleDateString();
}

// mobile-client/src/features/drive/utils/fileIcon.js

/**
 * Returns icon meta (Ionicons name + color) by file name/mime + folder flag
 */
export function getFileIconMeta(name, mimeType, isFolder) {
  if (isFolder) return { icon: "folder", color: "#f8d775" };

  const safeName = String(name || "");
  const ext = safeName.includes(".") ? safeName.split(".").pop().toLowerCase() : "";
  const mime = String(mimeType || "").toLowerCase();

  // Documents
  if (["doc", "docx", "odt"].includes(ext) || mime.includes("word"))
    return { icon: "document-text", color: "#4285f4" };

  // Spreadsheets
  if (["xls", "xlsx", "csv", "ods"].includes(ext) || mime.includes("spreadsheet") || mime.includes("excel"))
    return { icon: "grid", color: "#34a853" };

  // Presentations
  if (["ppt", "pptx", "odp"].includes(ext) || mime.includes("presentation") || mime.includes("powerpoint"))
    return { icon: "easel", color: "#f4b400" };

  // PDF
  if (ext === "pdf" || mime === "application/pdf")
    return { icon: "document", color: "#ea4335" };

  // Images
  if (["jpg", "jpeg", "png", "gif", "svg", "webp", "heic"].includes(ext) || mime.startsWith("image/"))
    return { icon: "image", color: "#ea4335" };

  // Video
  if (["mp4", "mkv", "avi", "mov"].includes(ext) || mime.startsWith("video/"))
    return { icon: "play-circle", color: "#ea4335" };

  // Audio
  if (["mp3", "wav", "ogg", "m4a"].includes(ext) || mime.startsWith("audio/"))
    return { icon: "musical-notes", color: "#a142f4" };

  // Archives
  if (["zip", "rar", "7z", "tar", "gz"].includes(ext) || mime.includes("zip") || mime.includes("compressed"))
    return { icon: "archive", color: "#fbbc04" };

  // Code
  if (["js", "jsx", "ts", "tsx", "html", "css", "json", "py", "java", "c", "cpp"].includes(ext))
    return { icon: "code-slash", color: "#4285f4" };

  // Text
  if (ext === "txt" || mime === "text/plain" || mime.startsWith("text/"))
    return { icon: "document-text", color: "#4285f4" };

  return { icon: "document-outline", color: "#9aa0a6" };
}

    pickerInProgressRef.current = true;

      setLoading(true);

      const base64 = await FileSystem.readAsStringAsync(uri, {
        encoding: "base64",
      });

      await createItem(token, {
        name: fileName,
        type: "file",
        content: base64,
        mimeType: mimeType || "application/octet-stream",
      });

      emitDriveRefresh({ reason: "upload", name: fileName });
      if (onRefresh) onRefresh();
    } catch (err) {

const API_BASE_URL = process.env.EXPO_PUBLIC_API_BASE_URL;

export async function request(path, options = {}) {
  const cleanPath = path.startsWith('/api') ? path.replace('/api', '') : path;
  const url = `${API_BASE_URL}${cleanPath}`;

  try {
    const isFormData = options.body instanceof FormData;

    const headers = {
      // ׳”׳•׳¡׳₪׳× JSON Content-Type ׳¨׳§ ׳׳ ׳–׳” ׳׳ FormData
      ...(isFormData ? {} : { "Content-Type": "application/json" }),
      ...(options.headers || {}),
    };

    if (options.token) {
      headers["Authorization"] = `Bearer ${options.token}`;
    }

    const res = await fetch(url, {
      ...options,
      headers,
      // ׳”׳׳¨׳× ׳׳•׳‘׳™׳™׳§׳˜ ׳-JSON ׳¨׳§ ׳׳ ׳”׳•׳ ׳׳•׳‘׳™׳™׳§׳˜ ׳•׳׳ FormData
      body: isFormData ? options.body : (options.body ? JSON.stringify(options.body) : undefined),
    });

    const text = await res.text();
    let data = null;
    try { data = text ? JSON.parse(text) : null; } catch { data = text; }

    if (!res.ok) {
      throw new Error(data?.error || data?.message || `Request failed with status ${res.status}`);
    }

    return data;
  } catch (error) {
    console.error("ג Request Error:", error.message);
    throw error;
  }
}

// src/shared_code/utils/formatBytes.js

export function formatBytes(bytes, decimals = 1) {
  const n = Number(bytes);
  if (!Number.isFinite(n) || n <= 0) return "0 B";

  const k = 1024;
  const sizes = ["B", "KB", "MB", "GB", "TB"];
  const i = Math.min(Math.floor(Math.log(n) / Math.log(k)), sizes.length - 1);

  const value = n / Math.pow(k, i);
  // ׳׳ ׳”׳׳¡׳₪׳¨ ׳©׳׳, ׳׳ ׳׳¦׳™׳’׳™׳ ׳ ׳§׳•׳“׳” ׳¢׳©׳¨׳•׳ ׳™׳×
  const fixed = Number.isInteger(value) ? 0 : decimals;
  
  return `${value.toFixed(fixed)} ${sizes[i]}`;
}
