const BASE_URL = "http://192.168.1.226:3000"; 

export async function request(path, options = {}) {
  const method = options.method ?? "GET";

  const headers = {
    "Content-Type": "application/json",
    ...(options.headers ?? {}),
  };

  if (options.token) headers.Authorization = `Bearer ${options.token}`;

  const res = await fetch(`${BASE_URL}${path}`, {
    method,
    headers,
    body: options.body ? JSON.stringify(options.body) : undefined,
  });

  const text = await res.text();
  const data = text ? safeJson(text) : null;

  if (!res.ok) {
    const msg =
      (data && (data.message || data.error)) ||
      `Request failed (${res.status})`;
    throw new Error(msg);
  }

  return data;
}

function safeJson(text) {
  try { return JSON.parse(text); } catch { return null; }
}

function getUserIdFromToken(token) {
  try {
    const base64Url = token.split('.')[1];
    const base64 = base64Url.replace(/-/g, '+').replace(/_/g, '/');
    const jsonPayload = decodeURIComponent(atob(base64).split('').map(function(c) {
        return '%' + ('00' + c.charCodeAt(0).toString(16)).slice(-2);
    }).join(''));

    return JSON.parse(jsonPayload).userId;
  } catch (e) {
    // Fallback ׳׳׳§׳¨׳” ׳©׳׳™׳ atob (׳›׳׳• ׳‘-Hermes engine)
    try {
        const payload = JSON.parse(String(Buffer.from(token.split('.')[1], 'base64')));
        return payload.userId;
    } catch (err) {
        console.log("Token decode error", err);
        return null; 
    }

const isText = (m = "", n = "") =>
  m.startsWith("text/") ||
  m === "application/json" ||
  m === "application/javascript" ||
  m === "application/xml" ||
  /\.(txt|md|csv|log|xml|js)$/i.test(n || "");

  m.startsWith("image/") || /\.(jpg|jpeg|png|gif|bmp|webp)$/i.test(n || "");

function getIcon(m = "", n = "") {
  if (isImage(m, n)) return "image-outline";
  if (isText(m, n)) return "document-text-outline";
  if (m.includes("pdf") || /\.pdf$/i.test(n || "")) return "document-outline";
  if (m.includes("zip") || /\.(zip|rar|7z|tar|gz)$/i.test(n || "")) return "archive-outline";
  if (m.startsWith("audio/") || /\.(mp3|wav|m4a|aac|flac|ogg)$/i.test(n || "")) return "musical-notes-outline";
  if (m.startsWith("video/") || /\.(mp4|mov|mkv|avi|webm)$/i.test(n || "")) return "videocam-outline";
  return "document-attach-outline";
}

// Robust JWT decode for RN/Expo
function decodeJwtSafe(token) {
  try {
    if (!token) return {};
    const base64Url = token.split(".")[1];
    if (!base64Url) return {};
    const base64 = base64Url.replace(/-/g, "+").replace(/_/g, "/");
    const padded = base64 + "===".slice((base64.length + 3) % 4);

    let jsonStr = "";
    if (typeof globalThis.atob === "function") {
      jsonStr = globalThis.atob(padded);
    } else {
      const { Buffer } = require("buffer");
      jsonStr = Buffer.from(padded, "base64").toString("utf8");
    }
    return JSON.parse(jsonStr);
  } catch {
    return {};
  }
}

    null
  );

import { useState, useRef } from "react";
import { Alert } from "react-native";
import * as ImagePicker from "expo-image-picker";
import * as DocumentPicker from "expo-document-picker";
import * as FileSystem from "expo-file-system/legacy";
import { createItem } from "../services/fileService";
import { emitDriveRefresh } from "../services/driveEvents";

export function useDriveUploads(token, onRefresh, closeCallback) {
  const [loading, setLoading] = useState(false);
  const pickerInProgressRef = useRef(false);

  const lockPicker = () => {
    if (pickerInProgressRef.current) return false;
    pickerInProgressRef.current = true;
    return true;
  };

  const unlockPicker = () => {
    pickerInProgressRef.current = false;
  };

  const processAndUpload = async (uri, fileName, mimeType) => {
    if (!token) {
      Alert.alert("Error", "No token found.");
      return;
    }

    try {
      setLoading(true);
      const base64 = await FileSystem.readAsStringAsync(uri, { encoding: "base64" });

      await createItem(token, {
        name: fileName,
        type: "file",
        content: base64,
        mimeType: mimeType || "application/octet-stream",
      });

      emitDriveRefresh({ reason: "upload", name: fileName });
      onRefresh?.();
    } catch (err) {
      Alert.alert("Upload Failed", err?.message || "Unknown error");
    } finally {
      setLoading(false);
      unlockPicker();
      closeCallback?.(); // ׳¡׳’׳™׳¨׳× ׳”׳׳•׳“׳ ׳‘׳¡׳™׳•׳
    }
  };

  const pickDocument = async () => {
    if (!lockPicker()) return;
    try {
      // ׳”׳©׳”׳™׳™׳” ׳§׳˜׳ ׳” ׳׳׳ ׳™׳¢׳× ׳”׳×׳ ׳’׳©׳•׳™׳•׳× UI
      await new Promise(r => setTimeout(r, 200));
      
      const result = await DocumentPicker.getDocumentAsync({
        copyToCacheDirectory: true,
        type: "*/*",
        multiple: false,
      });

      if (result?.canceled) {
        unlockPicker();
        return;
      }

      const asset = result?.assets?.[0];
      if (asset?.uri) {
        const fileName = asset.name || `file_${Date.now()}`;
        closeCallback?.(); // ׳¡׳•׳’׳¨׳™׳ ׳׳× ׳”׳×׳₪׳¨׳™׳˜ ׳׳₪׳ ׳™ ׳”׳”׳¢׳׳׳”
        await processAndUpload(asset.uri, fileName, asset.mimeType);
      }
    } catch (err) {
      Alert.alert("Error", "Picker failed");
      unlockPicker();
    }
  };

  const pickImage = async (useCamera) => {
    if (!lockPicker()) return;
    try {
      const permission = useCamera
        ? await ImagePicker.requestCameraPermissionsAsync()
        : await ImagePicker.requestMediaLibraryPermissionsAsync();

      if (!permission.granted) {
        Alert.alert("Permission Denied", "Settings required.");
        unlockPicker();
        return;
      }

      const options = {
        mediaTypes: ImagePicker.MediaTypeOptions.Images,
        quality: 0.7,
        allowsEditing: false,
      };

      const result = useCamera
        ? await ImagePicker.launchCameraAsync(options)
        : await ImagePicker.launchImageLibraryAsync(options);

      if (result?.canceled) {
        unlockPicker();
        return;
      }

      const asset = result?.assets?.[0];
      if (asset?.uri) {
        let fileName = asset.fileName;
        if (!fileName) {
          const uriParts = String(asset.uri).split("/");
          fileName = uriParts[uriParts.length - 1] || `photo_${Date.now()}.jpg`;
        }
        if (fileName && !fileName.includes(".")) fileName += ".jpg";
        
        closeCallback?.();
        await processAndUpload(asset.uri, fileName, asset.mimeType || "image/jpeg");
      }
    } catch (err) {
      Alert.alert("Error", "Action failed");
      unlockPicker();
    }
  };

  return { loading, pickDocument, pickImage };
}

// mobile-client/src/features/drive/hooks/useMyDrive.js

import { useState, useEffect, useCallback } from "react";
import { useAuth } from "../../../app/providers/AuthContext";
import { getFiles } from "../services/fileService";
import { subscribeDriveRefresh } from "../services/driveEvents";

export function useMyDrive() {
  const { token, logout } = useAuth();

  const [items, setItems] = useState([]);
  const [loading, setLoading] = useState(true);
  const [refreshing, setRefreshing] = useState(false);

  const fetchItems = useCallback(async () => {
    if (!token) return;

    try {
      const data = await getFiles(token);
      setItems(Array.isArray(data) ? data : []);
    } catch (error) {
      if (
        error?.message &&
        (error.message.includes("token") || error.message.includes("401"))
      ) {
        await logout();
      }
    } finally {
      setLoading(false);
      setRefreshing(false);
    }
  }, [token, logout]);

  useEffect(() => {
    fetchItems();
  }, [fetchItems]);

  // ג… listen for global refresh events (upload/create/delete/etc.)
  useEffect(() => {
    const unsubscribe = subscribeDriveRefresh(() => {
      setRefreshing(true);
      fetchItems();
    });
    return unsubscribe;
  }, [fetchItems]);

  const onRefresh = () => {
    setRefreshing(true);
    fetchItems();
  };

  return { items, loading, refreshing, onRefresh };
}

  if (!trashedAt) return "";
  const trashDate = new Date(trashedAt);
  const deleteDate = new Date(trashDate.getTime() + 30 * 24 * 60 * 60 * 1000);
  const now = new Date();
  const diffTime = deleteDate - now;
  const diffDays = Math.ceil(diffTime / (1000 * 60 * 60 * 24));

}

export function formatLastAccessed(dateString) {
  if (!dateString) return "Never";
  const date = new Date(dateString);
  const diffMs = new Date() - date;
  const diffMins = Math.floor(diffMs / 60000);

  if (diffMins < 1) return "Just now";
  if (diffMins < 60) return `${diffMins} mins ago`;
  if (diffMs < 86400000) return `${Math.floor(diffMs / 3600000)} hours ago`;
  return date.toLocaleDateString();
}

import * as FileSystem from "expo-file-system";
import * as Sharing from "expo-sharing";

export async function downloadAndShareBase64({ base64, mimeType, fileName }) {
  if (!base64) throw new Error("No content to download");

  const safeName = (fileName || "file").replace(/[\/\\?%*:|"<>]/g, "_");
  const uri = FileSystem.cacheDirectory + safeName;

  await FileSystem.writeAsStringAsync(uri, base64, {
    encoding: FileSystem.EncodingType.Base64,
  });

  const canShare = await Sharing.isAvailableAsync();
  if (canShare) {
    await Sharing.shareAsync(uri, {
      mimeType: mimeType || "application/octet-stream",
      dialogTitle: "Download / Open file",
    });
  }

  return uri;
}

// mobile-client/src/features/drive/utils/fileIcon.js

/**
 * Returns icon meta (Ionicons name + color) by file name/mime + folder flag
 */
export function getFileIconMeta(name, mimeType, isFolder) {
  if (isFolder) return { icon: "folder", color: "#f8d775" };

  const safeName = String(name || "");
  const ext = safeName.includes(".") ? safeName.split(".").pop().toLowerCase() : "";
  const mime = String(mimeType || "").toLowerCase();

  // Documents
  if (["doc", "docx", "odt"].includes(ext) || mime.includes("word"))
    return { icon: "document-text", color: "#4285f4" };

  // Spreadsheets
  if (["xls", "xlsx", "csv", "ods"].includes(ext) || mime.includes("spreadsheet") || mime.includes("excel"))
    return { icon: "grid", color: "#34a853" };

  // Presentations
  if (["ppt", "pptx", "odp"].includes(ext) || mime.includes("presentation") || mime.includes("powerpoint"))
    return { icon: "easel", color: "#f4b400" };

  // PDF
  if (ext === "pdf" || mime === "application/pdf")
    return { icon: "document", color: "#ea4335" };

  // Images
  if (["jpg", "jpeg", "png", "gif", "svg", "webp", "heic"].includes(ext) || mime.startsWith("image/"))
    return { icon: "image", color: "#ea4335" };

  // Video
  if (["mp4", "mkv", "avi", "mov"].includes(ext) || mime.startsWith("video/"))
    return { icon: "play-circle", color: "#ea4335" };

  // Audio
  if (["mp3", "wav", "ogg", "m4a"].includes(ext) || mime.startsWith("audio/"))
    return { icon: "musical-notes", color: "#a142f4" };

  // Archives
  if (["zip", "rar", "7z", "tar", "gz"].includes(ext) || mime.includes("zip") || mime.includes("compressed"))
    return { icon: "archive", color: "#fbbc04" };

  // Code
  if (["js", "jsx", "ts", "tsx", "html", "css", "json", "py", "java", "c", "cpp"].includes(ext))
    return { icon: "code-slash", color: "#4285f4" };

  // Text
  if (ext === "txt" || mime === "text/plain" || mime.startsWith("text/"))
    return { icon: "document-text", color: "#4285f4" };

  return { icon: "document-outline", color: "#9aa0a6" };
}

  const n = (name || "").toLowerCase();
  return (
    mime === "text/plain" ||
    mime.startsWith("text/") ||
    mime === "application/json" ||
    mime === "application/javascript" ||
    mime === "application/xml" ||
    n.endsWith(".txt")
  );
};

export const isImageFile = (mime = "", name = "") => {
  const n = (name || "").toLowerCase();
  return (
    (mime || "").startsWith("image/") ||
    /\.(jpg|jpeg|png|gif|bmp|webp)$/i.test(n)
  );
};

  const n = (name || "").toLowerCase();
  return mime === "application/pdf" || n.endsWith(".pdf");
};

export const isVideoFile = (mime = "", name = "") => {
  const n = (name || "").toLowerCase();
  return (mime || "").startsWith("video/") || /\.(mp4|mov|mkv|webm)$/i.test(n);
};

    const targetX = (index * TAB_WIDTH) + BUBBLE_OFFSET;

      speed: 15,

  const [loading, setLoading] = useState(false);
  const [parentId, setParentId] = useState(null);
  const [showFolderPicker, setShowFolderPicker] = useState(false);

  const pickerInProgressRef = useRef(false);

  // Get all folders for the parent dropdown
  const folders = useMemo(() => {
    return (items || []).filter((item) => item.type === "folder");
  }, [items]);

    if (!parentId) return "Root (My Drive)";

      setLoading(true);

      const base64 = await FileSystem.readAsStringAsync(uri, {
        encoding: "base64",
      });

      await createItem(token, {
        name: fileName,
        type: "file",
        content: base64,
        mimeType: mimeType || "application/octet-stream",
      });

      emitDriveRefresh({ reason: "upload", name: fileName });
      onRefresh?.();
    } catch (err) {
      Alert.alert("Upload Failed", err?.message || "Unknown error");

      setLoading(false);
      unlockPicker();

      const fileName = asset.name || `file_${Date.now()}`;
      await processAndUpload(asset.uri, fileName, asset.mimeType);

export async function request(path, options = {}) {
  // Ensure path starts with /
  const normalizedPath = path.startsWith("/") ? path : `/${path}`;
  
  // Build the full URL
  const url = `${API_BASE_URL}${normalizedPath}`;
  
  console.log("נ“¡ Request:", options.method || "GET", url);

  try {
    const isFormData = options.body instanceof FormData;

    const headers = {
      ...(isFormData ? {} : { "Content-Type": "application/json" }),
      ...(options.headers || {}),
    };

    if (options.token) {
      headers["Authorization"] = `Bearer ${options.token}`;
    }

    const res = await fetch(url, {
      ...options,
      headers,
      body: isFormData ? options.body : (options.body ? JSON.stringify(options.body) : undefined),
    });

    const text = await res.text();
    let data = null;
    try {
      data = text ? JSON.parse(text) : null;
    } catch {
      data = text;
    }

    if (!res.ok) {
      const msg =
        (data && (data.message || data.error)) ||
        (typeof data === "string" && data) ||
        `Request failed (${res.status})`;
      throw new Error(msg);
    }

    return data;
  } catch (error) {
    console.error("Request Error:", error?.message || error);
    throw error;
  }
}

// src/shared_code/utils/formatBytes.js

export function formatBytes(bytes, decimals = 1) {
  const n = Number(bytes);
  if (!Number.isFinite(n) || n <= 0) return "0 B";

  const k = 1024;
  const sizes = ["B", "KB", "MB", "GB", "TB"];
  const i = Math.min(Math.floor(Math.log(n) / Math.log(k)), sizes.length - 1);

  const value = n / Math.pow(k, i);
  // ׳׳ ׳”׳׳¡׳₪׳¨ ׳©׳׳, ׳׳ ׳׳¦׳™׳’׳™׳ ׳ ׳§׳•׳“׳” ׳¢׳©׳¨׳•׳ ׳™׳×
  const fixed = Number.isInteger(value) ? 0 : decimals;
  
  return `${value.toFixed(fixed)} ${sizes[i]}`;
}
