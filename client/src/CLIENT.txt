let unauthorizedHandler = null;

export function setUnauthorizedHandler(handler) {
  unauthorizedHandler = handler;

function getAuthToken() {
  return localStorage.getItem("token");
}

const BASE_URL = "http://localhost:3000";

/**
 * Unified request helper:
 * - Adds Authorization header
 * - Triggers unauthorizedHandler on 401
 * - Safely parses JSON / empty bodies
 * - Returns: parsed JSON, or null for 204/empty body
 */
export async function request(path, options = {}) {
  const token = getAuthToken();
  if (!token) throw new Error("Not authenticated");

  const url = path.startsWith("http") ? path : `${BASE_URL}${path}`;

  const res = await fetch(url, {
    ...options,
    headers: {
      Authorization: `Bearer ${token}`,
      ...(options.headers || {}),
    },
  });

  if (res.status === 401 && unauthorizedHandler) {
    unauthorizedHandler(); // triggers Modal in App.js
  }

  // No Content
  if (res.status === 204) return null;

  // Try read body (may be empty)
  const text = await res.text();
  const data = text
    ? (() => {
      try {
        return JSON.parse(text);
      } catch {
        return text;
      }
    })()
    : null;

  if (!res.ok) {
    // backend might return {error} or {message}
    const msg =
      (data && typeof data === "object" && (data.error || data.message)) ||
      "Request failed";
    throw new Error(msg);
  }

  return data;
}

export async function openItem(item, onFolderOpen = null) {
  if (!item) throw new Error("No item provided");

  if (item.type === "folder") {
    if (onFolderOpen) onFolderOpen(item);
    return null;
  }

  // fire-and-forget update access time
  request(`/api/files/${item.id}/access`, {
    method: "PATCH",
    headers: { "Content-Type": "application/json" },
  }).catch(() => { });

  return getFileById(item.id);
}

export function formatLastAccessed(dateString) {
  if (!dateString) return "Never";
  const date = new Date(dateString);
  const diffMs = new Date() - date;
  const diffMins = Math.floor(diffMs / 60000);
  if (diffMins < 1) return "Just now";
  if (diffMins < 60) return `${diffMins} mins ago`;
  if (diffMs < 86400000) return `${Math.floor(diffMs / 3600000)} hours ago`;
  return date.toLocaleDateString();
}

export async function request(path, options = {}) {
    const token = getToken();

    const headers = {
        ...(options.headers || {}),
    };

    // ׳׳ ׳׳™׳ header Authorization ׳•׳¢׳“׳™׳™׳ ׳™׳© token - ׳ ׳•׳¡׳™׳£
    if (token && !headers.Authorization) {
        headers.Authorization = `Bearer ${token}`;
    }

    const res = await fetch(path, {
        ...options,
        headers,
    });

    // ׳˜׳™׳₪׳•׳ ׳‘-401 ׳׳—׳™׳“
    if (res.status === 401) {
        if (typeof unauthorizedHandler === "function") unauthorizedHandler();
        const text = await res.text().catch(() => "");
        throw new Error(text || "Unauthorized");
    }

    // ׳©׳’׳™׳׳•׳× ׳›׳׳׳™׳•׳×
    if (!res.ok) {
        const text = await res.text().catch(() => "");
        throw new Error(text || `Request failed (${res.status})`);
    }

    // ׳׳ ׳×׳׳™׳“ ׳™׳© ׳’׳•׳£ (204 ׳•׳›׳•׳³)
    const contentType = res.headers.get("content-type") || "";
    if (contentType.includes("application/json")) return res.json();
    return res.text();
}

const shallowEqual = (a, b) => {
  const ak = Object.keys(a), bk = Object.keys(b);

};

export function AuthProvider({ children }) {
  const [token, setToken] = useState(() => {
    return localStorage.getItem("token");
  });

  // ג… ׳—׳“׳©: ׳׳¦׳‘ ׳’׳׳•׳‘׳׳™ ׳׳¡׳©׳ ׳©׳₪׳’ ׳×׳•׳§׳£
  const [sessionExpired, setSessionExpired] = useState(false);

  useEffect(() => {
    const storedToken = localStorage.getItem("token");
    if (storedToken !== token) {
      setToken(storedToken);
    }
  }, []);

  const login = (newToken) => {
    localStorage.setItem("token", newToken);
    setToken(newToken);
    setSessionExpired(false); // ׳׳ ׳”׳×׳—׳‘׳¨׳ ׳• ׳׳—׳“׳© ג€“ ׳׳™׳ ׳¡׳©׳ ׳©׳₪׳’
  };

  const logout = () => {
    localStorage.removeItem("token");
    setToken(null);
    setSessionExpired(false);
  };

  // ג… ׳—׳“׳©: ׳׳₪׳¢׳™׳ ׳׳× ׳”׳׳•׳“׳׳ (׳׳ ׳׳×׳ ׳×׳§ ׳׳™׳“)
  const triggerSessionExpired = () => {
    setSessionExpired(true);
  };

  const isAuthenticated = !!token && token !== "undefined";

  const value = useMemo(
    () => ({
      token,
      login,
      logout,
      isAuthenticated,

      // ג… ׳—׳“׳©
      sessionExpired,
      triggerSessionExpired,
    }),
    [token, isAuthenticated, sessionExpired]
  );

  return (
    <AuthContext.Provider value={value}>
      {children}
    </AuthContext.Provider>
  );
}

  const context = useContext(AuthContext);
  if (!context) {
    throw new Error("useAuth must be used within AuthProvider");
  }

 * Authentication Service
 * Handles all authentication-related API calls

const API_BASE_URL = 'http://localhost:3000/api';

export const authenticateUser = async (email, password) => {
    try {
        const response = await fetch(`${API_BASE_URL}/tokens`, {
            method: 'POST',
            headers: { 'Content-Type': 'application/json' },
            body: JSON.stringify({ email, password })
        });

        const data = await response.json().catch(() => ({}));

        if (response.ok) {
            return { success: true, token: data.token, error: null };
        }

        if (response.status === 404) {
            return { success: false, error: "User not found. Please check your email or create an account." };
        }

        if (response.status === 401) {
            return { success: false, error: "Incorrect password. Please try again." };
        }

        return { success: false, error: data.error || "An error occurred on the server." };

    } catch (error) {
        console.error("Authentication error:", error);
        return { success: false, error: "Server connection error. Ensure the backend is running." };
    }
};

// client/src/utils/formatBytes.js
export function formatBytes(bytes, decimals = 1) {
  const n = Number(bytes);
  if (!Number.isFinite(n) || n <= 0) return "0 B";

  const k = 1024;
  const sizes = ["B", "KB", "MB", "GB", "TB"];
  const i = Math.min(Math.floor(Math.log(n) / Math.log(k)), sizes.length - 1);

  const value = n / Math.pow(k, i);
  const fixed = i === 0 ? 0 : decimals;
  return `${value.toFixed(fixed)} ${sizes[i]}`;
}

/**
 * Generates a default profile picture based on user's initials
 * @param {string} firstName - User's first name
 * @param {string} lastName - User's last name (optional)
 * @returns {string} - SVG data URL for default profile picture
 */
export const generateDefaultProfilePicture = (firstName, lastName = '') => {
    // Get initials (max 2 characters)
    const initials = (firstName.charAt(0) + (lastName ? lastName.charAt(0) : '')).toUpperCase();

    // Generate a consistent color based on the name
    const colors = [
        '#1a73e8', // Copa Blue
        '#34a853', // Copa Green
        '#ea4335', // Copa Red
        '#fbbc04', // Copa Yellow
        '#ff6d00', // Orange
        '#9334e6', // Purple
    ];

    const colorIndex = (firstName.charCodeAt(0) + (lastName ? lastName.charCodeAt(0) : 0)) % colors.length;
    const backgroundColor = colors[colorIndex];

    // Create SVG
    const svg = `
        <svg width="120" height="120" xmlns="http://www.w3.org/2000/svg">
            <circle cx="60" cy="60" r="60" fill="${backgroundColor}"/>
            <text x="50%" y="50%" dominant-baseline="middle" text-anchor="middle" 
                  font-family="Roboto, Arial, sans-serif" font-size="48" font-weight="400" fill="white">
                ${initials}
            </text>
        </svg>
    `;

    // Convert SVG to base64 data URL
    return `data:image/svg+xml;base64,${btoa(svg)}`;
};

/**
 * Gets the profile picture - either uploaded or default
 * @param {File|null} uploadedImage - The uploaded image file
 * @param {string} firstName - User's first name
 * @param {string} lastName - User's last name (optional)
 * @returns {Promise<string>} - Base64 string of the profile picture
 */
export const getProfilePicture = async (uploadedImage, firstName, lastName = '') => {
    if (uploadedImage) {
        // If user uploaded an image, convert it to base64
        return new Promise((resolve, reject) => {
            const reader = new FileReader();
            reader.readAsDataURL(uploadedImage);
            reader.onload = () => resolve(reader.result);
            reader.onerror = (error) => reject(error);
        });
    } else {
        // If no image uploaded, return default profile picture
        return generateDefaultProfilePicture(firstName, lastName);
    }
};

/**
 * Validates if the uploaded image is acceptable
 * @param {File} file - The image file
 * @returns {Object} - { valid: boolean, error: string|null }
 */
export const validateProfilePicture = (file) => {
    if (!file) {
        return { valid: true, error: null }; // No file is okay
    }

    // Check file size (max 5MB)
    if (file.size > 5000000) {
        return { valid: false, error: "Image size must be less than 5MB" };
    }

    // Check file type
    const validTypes = ['image/jpeg', 'image/jpg', 'image/png', 'image/gif', 'image/webp'];
    if (!validTypes.includes(file.type)) {
        return { valid: false, error: "Please upload a valid image file (JPEG, PNG, GIF, or WebP)" };
    }

    return { valid: true, error: null };
};
