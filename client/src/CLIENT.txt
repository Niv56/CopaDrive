import React, {
  createContext,
  useState,
  useContext,
  useEffect,
  useMemo
} from "react";

const AuthContext = createContext();

export function AuthProvider({ children }) {
  // Initialize token from localStorage
  const [token, setToken] = useState(() => {
    return localStorage.getItem("token");
  });

  // Global state to track if the session has expired (e.g., 401 response)
  const [sessionExpired, setSessionExpired] = useState(false);

  // Sync state if localStorage changes externally (optional safeguard)
  useEffect(() => {
    const storedToken = localStorage.getItem("token");
    if (storedToken !== token) {
      setToken(storedToken);
    }
  }, []);

  const login = (newToken) => {
    localStorage.setItem("token", newToken);
    setToken(newToken);
    setSessionExpired(false); // Reset expiration state on fresh login
  };

  const logout = () => {
    localStorage.removeItem("token");
    setToken(null);
    setSessionExpired(false);
  };

  // Trigger the "Session Expired" modal without immediately logging out the user
  // This allows the user to see *why* they are being asked to log in again
  const triggerSessionExpired = () => {
    setSessionExpired(true);
  };

  const isAuthenticated = !!token && token !== "undefined";

  const value = useMemo(
    () => ({
      token,
      login,
      logout,
      isAuthenticated,
      sessionExpired,
      triggerSessionExpired,
    }),
    [token, isAuthenticated, sessionExpired]
  );

  return (
    <AuthContext.Provider value={value}>
      {children}
    </AuthContext.Provider>
  );
}

export function useAuth() {
  const context = useContext(AuthContext);
  if (!context) {
    throw new Error("useAuth must be used within AuthProvider");
  }
  return context;
}

import React, { createContext, useContext, useEffect, useState } from "react";

const ThemeContext = createContext();

export const useTheme = () => useContext(ThemeContext);

export const ThemeProvider = ({ children }) => {
  // Load saved theme preference, defaulting to "light" when missing.
  const [theme, setTheme] = useState(() => localStorage.getItem("app-theme") || "light");

  useEffect(() => {
    // Apply theme attribute on the root element so CSS can react to it.
    document.documentElement.setAttribute("data-theme", theme);
    localStorage.setItem("app-theme", theme);
  }, [theme]);

  // Toggle between "light" and "dark" themes.
  const toggleTheme = () => setTheme((prev) => (prev === "light" ? "dark" : "light"));

  return (
    <ThemeContext.Provider value={{ theme, toggleTheme }}>
      {children}
    </ThemeContext.Provider>
  );
};

const NEWLINE_PLACEHOLDER = "_@NL@_";


function TextViewer({ file }) {
  const [content, setContent] = useState(null);

  useEffect(() => {
    if (!file) return;

    if (file.sizeBytes === 0 || !file.content) {
      setContent(""); 
      return;
    }

    
    const processed = file.content.replaceAll(NEWLINE_PLACEHOLDER, "\n");
    
    console.log(`[TEXT-VIEWER] Formatting file: ${file.name}`);
    setContent(processed);
  }, [file]);

  if (content === null) {
    return <div className="p-3 text-center text-muted">Loading content...</div>;
  }

  return (
    <div className="text-viewer-container p-3">
      {content.trim() === "" ? (
        <div className="no-content-message p-5 text-center text-muted border rounded bg-light">
           File is empty
        </div>
      ) : (
        <textarea
          className="form-control"
          rows="15"

    if (!match) return { y: "", m: "", d: "" };

  };

  useEffect(() => {
    const match = String(formData.dateOfBirth || "").match(/^\d{4}-\d{2}-\d{2}$/);
    if (match) setDob(parseDob(formData.dateOfBirth));
  }, [formData.dateOfBirth]);

  const months = useMemo(
    () => [
      { value: "01", label: "Jan" }, { value: "02", label: "Feb" },
      { value: "03", label: "Mar" }, { value: "04", label: "Apr" },
      { value: "05", label: "May" }, { value: "06", label: "Jun" },
      { value: "07", label: "Jul" }, { value: "08", label: "Aug" },
      { value: "09", label: "Sep" }, { value: "10", label: "Oct" },
      { value: "11", label: "Nov" }, { value: "12", label: "Dec" },
    ],
    []
  );

  // Year list: Generates years allowing ages between 10 and 120
  const years = useMemo(() => {
    const currentYear = new Date().getFullYear();
    const maxYear = currentYear - 10; // Youngest allowed (e.g., 2016)
    const minYear = currentYear - 120; // Oldest allowed
    const arr = [];
    for (let y = maxYear; y >= minYear; y--) arr.push(String(y));
    return arr;
  }, []);

  // Calculate days in a specific month/year
  const daysInMonth = (year, month) => {

    const mm = Number(month);

  const res = await fetch(url, {

    ? (() => {

          return JSON.parse(text);

          return text;

      })()
    : null;

  if (!res.ok) {
    const msg = (data && typeof data === "object" && (data.error || data.message)) || "Request failed";
    throw new Error(msg);

import { useEffect, useRef, useState } from "react";
import { getCurrentUser } from "features/drive/api/filesApi";

// Build a cache of ownerId -> user and keep it updated during the session.
export function useOwnersMap(items) {
  const [owners, setOwners] = useState({});
  const fetchedRef = useRef(new Set());

  // Fetch missing owners only once per session.
  useEffect(() => {
    const ids = Array.from(new Set((items || []).map((x) => x?.ownerId).filter(Boolean)));

    ids.forEach((id) => {
      if (fetchedRef.current.has(id)) return;
      fetchedRef.current.add(id);

      getCurrentUser(id)
        .then((user) => {
          setOwners((prev) => ({ ...prev, [id]: user }));
        })
        .catch(() => {
          // Ignore fetch errors and fall back to default UI.
        });
    });
  }, [items]);

  // Listen for "user-updated" events and update the cache immediately.
  useEffect(() => {
    const handler = (e) => {
      const { userId, user } = e.detail || {};
      if (!userId || !user) return;

      setOwners((prev) => ({ ...prev, [userId]: user }));
      fetchedRef.current.add(userId);
    };

    window.addEventListener("user-updated", handler);
    return () => window.removeEventListener("user-updated", handler);
  }, []);

  return owners;
}

    const [foundUser, setFoundUser] = useState(null);
    const [loading, setLoading] = useState(false);
    const [error, setError] = useState("");

    const displayName = useMemo(() => {
        if (!foundUser) return "";
        const full = `${foundUser.firstName || ""} ${foundUser.lastName || ""}`.trim();
        return full || foundUser.email?.split("@")[0] || "";
    }, [foundUser]);

        const raw = foundUser.image;

        setLoading(true);

import React, { createContext, useContext, useEffect, useMemo, useState } from "react";

const ViewModeContext = createContext(null);

const STORAGE_KEY = "copadrive:viewMode"; // "list" | "grid"

// Provider component to manage list/grid view state across the app
export function ViewModeProvider({ children }) {
  // Initialize state from localStorage (default to "list")
  const [viewMode, setViewMode] = useState(() => {
    const saved = localStorage.getItem(STORAGE_KEY);
    return saved === "grid" ? "grid" : "list";
  });

  // Persist view mode changes to localStorage
  useEffect(() => {
    localStorage.setItem(STORAGE_KEY, viewMode);
  }, [viewMode]);

  const value = useMemo(() => ({ viewMode, setViewMode }), [viewMode]);

  return <ViewModeContext.Provider value={value}>{children}</ViewModeContext.Provider>;
}

// Custom hook to access view mode context
export function useViewMode() {
  const ctx = useContext(ViewModeContext);
  if (!ctx) throw new Error("useViewMode must be used inside <ViewModeProvider />");
  return ctx;
}

import React, { useRef, useImperativeHandle, forwardRef } from "react";

const FileUploadHandler = forwardRef(({ onFileProcessed }, ref) => {
    const fileInputRef = useRef(null);

    // Expose trigger function to parent with optional MIME filter
    useImperativeHandle(ref, () => ({
        triggerUpload: (filter = "*/*") => {
            if (fileInputRef.current) {
                fileInputRef.current.accept = filter;
                fileInputRef.current.click();
            }
        }
    }));

    const handleFileChange = (e) => {
        const file = e.target.files[0];
        if (!file) return;

        const reader = new FileReader();
        const isTextFile = file.type === "text/plain" || file.name.endsWith(".txt");

        reader.onload = () => {
            const result = reader.result;
            
            // Safety check for empty or failed reads
            if (result === null) {
                console.error("FileReader result is null");
                return;
            }

            let finalContent = "";

            if (isTextFile) {
                // Keep text content as-is (even if empty string)
                finalContent = result;
            } else {
                // Extract Base64 string for binary files
                const base64Parts = result.split(',');
                finalContent = base64Parts.length > 1 ? base64Parts[1] : result;
            }

            // Send processed data to the parent component
            onFileProcessed({
                name: file.name,
                content: finalContent,
                mimeType: file.type || (isTextFile ? "text/plain" : "application/octet-stream"),
                size: file.size,
                isText: isTextFile
            });
        };

        reader.onerror = (err) => {
            console.error("FileReader error:", err);
            alert("Failed to read file from local disk.");
        };

        // Choose reading method based on file type
        if (isTextFile) {
            reader.readAsText(file);
        } else {
            reader.readAsDataURL(file);
        }

        // Reset input to allow selecting the same file again
        e.target.value = "";
    };

    return (
        <input
            type="file"
            ref={fileInputRef}
            style={{ display: "none" }}
            onChange={handleFileChange}
        />
    );
});

export const encodeTextToBase64 = (str) => {
  if (!str) return "";
  try {
    const bytes = new TextEncoder().encode(str);
    // ׳©׳™׳׳•׳© ׳‘-fromCodePoint/fromCharCode ׳‘׳¦׳•׳¨׳” ׳׳¨׳•׳›׳–׳× ׳׳™׳¢׳™׳׳•׳×
    const binString = String.fromCodePoint(...bytes);
    return btoa(binString);
  } catch (e) {
    console.error("Encoding error:", e);
    return "";
  }
};

export const decodeBase64ToText = (base64) => {
  if (!base64) return "";
  try {
    const binString = atob(base64);
    const bytes = Uint8Array.from(binString, (m) => m.codePointAt(0));
    return new TextDecoder().decode(bytes);
  } catch (e) {
    console.error("Decoding error:", e);
    return "Error: Could not decode text content.";
  }
};

export const base64ToBlob = (base64, mimeType = "application/octet-stream") => {
  try {
    // ׳ ׳™׳§׳•׳™ ׳׳₪׳©׳¨׳™ ׳©׳ ׳”׳§׳™׳“׳•׳׳× ׳‘׳׳§׳¨׳” ׳©׳ ׳©׳׳— Data URL ׳׳׳
    const cleanBase64 = base64.includes(",") ? base64.split(",")[1] : base64;
    
    const binString = atob(cleanBase64);
    const bytes = Uint8Array.from(binString, (m) => m.codePointAt(0));
    return new Blob([bytes], { type: mimeType });
  } catch (e) {
    console.error("Blob conversion error:", e);
    return null;
  }
};

export const fileToBase64 = (file) => {
  return new Promise((resolve, reject) => {
    if (!(file instanceof File || file instanceof Blob)) {
      reject(new Error("Input is not a file"));
      return;
    }
    const reader = new FileReader();
    reader.readAsDataURL(file);
    reader.onload = () => resolve(reader.result.split(",")[1]);
    reader.onerror = (error) => reject(error);
  });
};

// Format a byte count into a human-readable string (B/KB/MB/GB/TB).
export function formatBytes(bytes, decimals = 1) {
  const n = Number(bytes);
  if (!Number.isFinite(n) || n <= 0) return "0 B";

  const k = 1024;
  const sizes = ["B", "KB", "MB", "GB", "TB"];
  const i = Math.min(Math.floor(Math.log(n) / Math.log(k)), sizes.length - 1);

  const value = n / Math.pow(k, i);
  const fixed = i === 0 ? 0 : decimals;
  return `${value.toFixed(fixed)} ${sizes[i]}`;
}

// Generates a default SVG profile picture based on user initials
export const generateDefaultProfilePicture = (firstName, lastName = '') => {
    const initials = (firstName.charAt(0) + (lastName ? lastName.charAt(0) : '')).toUpperCase();

    const colors = [
        '#1a73e8', // Copa Blue
        '#34a853', // Copa Green
        '#ea4335', // Copa Red
        '#fbbc04', // Copa Yellow
        '#ff6d00', // Orange
        '#9334e6', // Purple
    ];

    const colorIndex = (firstName.charCodeAt(0) + (lastName ? lastName.charCodeAt(0) : 0)) % colors.length;
    const backgroundColor = colors[colorIndex];

    const svg = `
        <svg width="120" height="120" xmlns="http://www.w3.org/2000/svg">
            <circle cx="60" cy="60" r="60" fill="${backgroundColor}"/>
            <text x="50%" y="50%" dominant-baseline="middle" text-anchor="middle" 
                  font-family="Roboto, Arial, sans-serif" font-size="48" font-weight="400" fill="white">
                ${initials}
            </text>
        </svg>
    `;

    return `data:image/svg+xml;base64,${btoa(svg)}`;
};

// Returns the uploaded image as Base64 or generates a default one if none provided
export const getProfilePicture = async (uploadedImage, firstName, lastName = '') => {
    if (uploadedImage) {
        return new Promise((resolve, reject) => {
            const reader = new FileReader();
            reader.readAsDataURL(uploadedImage);
            reader.onload = () => resolve(reader.result);
            reader.onerror = (error) => reject(error);
        });
    } else {
        return generateDefaultProfilePicture(firstName, lastName);
    }
};

// Validates image file size (max 5MB) and allowed file types
export const validateProfilePicture = (file) => {
    if (!file) {
        return { valid: true, error: null };
    }

    if (file.size > 5000000) {
        return { valid: false, error: "Image size must be less than 5MB" };
    }

    const validTypes = ['image/jpeg', 'image/jpg', 'image/png', 'image/gif', 'image/webp'];
    if (!validTypes.includes(file.type)) {
        return { valid: false, error: "Please upload a valid image file (JPEG, PNG, GIF, or WebP)" };
    }

    return { valid: true, error: null };
};
