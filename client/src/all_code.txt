function getAuthToken() {
  return localStorage.getItem("accessToken");
}

export async function getRootFiles() {
  const token = getAuthToken();
  if (!token) throw new Error("Not authenticated");



//Get api/files - all files in top level
  const response = await fetch("/api/files", {
    method: "GET",
    headers: { Authorization: `Bearer ${token}` },
  });

  if (!response.ok) throw new Error("Failed to fetch files");
  return response.json();
}


// Post api/files 
export async function createItem({ name, type, parentId = null, content = "" }) {
  const token = getAuthToken();
  if (!token) throw new Error("Not authenticated");

  const body = {
    name,
    type,                 
    parentId,            
    content: type === "file" ? content : "", 
  };

  const response = await fetch("/api/files", {
    method: "POST",
    headers: {
      Authorization: `Bearer ${token}`,
      "Content-Type": "application/json",
    },
    body: JSON.stringify(body),
  });

  if (!response.ok) {
    throw new Error("Failed to create item");
  }

  return response.json(); 
}

const BASE_URL = import.meta.env.VITE_API_BASE_URL || "http://localhost:3000";

export async function http(path, options = {}) {
  const url = `${BASE_URL}${path}`;

  const headers = {
    "Content-Type": "application/json",
    ...(options.headers || {}),
  };

  const token = localStorage.getItem("token");
  if (token) {
    headers.Authorization = `Bearer ${token}`;
  }

  const res = await fetch(url, {
    ...options,
    headers,
  });

  if (!res.ok) {
    let errorBody = null;
    try {
      errorBody = await res.json();
    } catch {
      // ignore
    }
    const err = new Error(`HTTP ${res.status}`);
    err.status = res.status;
    err.body = errorBody;
    throw err;
  }

  // ׳׳ ׳׳™׳ ׳×׳•׳›׳ (204), ׳ ׳—׳–׳™׳¨ null
  if (res.status === 204) return null;

  // ׳‘׳¨׳™׳¨׳× ׳׳—׳“׳: JSON
  return res.json();
}

import FileViewerModal from "../widgets/FileViewerModal";
import "./MyDrive.css";

function MyDrive() {
  const [allFiles, setAllFiles] = useState([]);
  const [loading, setLoading] = useState(true);
  const [error, setError] = useState(null);

  const [currentFolderId, setCurrentFolderId] = useState(null);
  const [openedFile, setOpenedFile] = useState(null);

  useEffect(() => {
    (async () => {
      try {
        setLoading(true);
        setError(null);

        const data = await getRootFiles(); // listTopLevel
        const list = Array.isArray(data) ? data : [];
        setAllFiles(list);
      } catch (e) {
        setError(e?.message ?? "Failed to load files");
      } finally {
        setLoading(false);
      }
    })();
  }, []);

  const currentItems = useMemo(() => {
    return allFiles.filter((f) => f.parentId === currentFolderId);
  }, [allFiles, currentFolderId]);

  const folders = useMemo(() => {
    return allFiles.filter((f) => f.type === "folder");
  }, [allFiles]);

  function onOpen(item) {
    if (item.type === "folder") {
      setCurrentFolderId(item.id);
    } else {
      setOpenedFile(item);
    }
  }

  if (loading) return <div>Loading...</div>;
  if (error) return <div style={{ color: "red" }}>{error}</div>;

  return (
    <div className="mainarea">
      <h2>My Drive</h2>

      <div className="files-container">
        {currentItems.map((item) => (
          <FileListItem
            key={item.id}
            item={item}
            onOpen={onOpen}
            onEdit={() => alert("׳™׳© ׳׳׳׳©")}
            onShare={() => alert("׳™׳© ׳׳׳׳©")}

import logo from "../../photos/logo.svg";

const STEP_CONFIG = {
  1: {
    title: "Create a Copa Account",
    subtitle: "Enter your name",
    validate: validateStep1,
  },
  2: {
    title: "Basic information",
    subtitle: "Add a profile picture and phone number",
    validate: validateStep2,
  },
  3: {
    title: "Basic information",
    subtitle: "Enter your date of birth and gender",

  },

    title: "Create your Copa Account",
    subtitle: "Enter your email and create a password",
    validate: validateStep4,
  },

const TOTAL_STEPS = 4;
const API_URL = "http://localhost:3000/api/users";

const Register = () => {
  const navigate = useNavigate();
  const fileInputRef = useRef(null);

  const [step, setStep] = useState(1);
  const [preview, setPreview] = useState(null);
  const [errors, setErrors] = useState({});

  const [formData, setFormData] = useState({
    firstName: "",
    lastName: "",
    phoneNumber: "",
    image: null,

    dateOfBirth: "",
    gender: "",

    email: "",
    password: "",
    verifyPassword: "",
  });

  const validateCurrentStep = () => {
    const stepErrors = STEP_CONFIG[step].validate(formData);
    setErrors(stepErrors);
    return Object.keys(stepErrors).length === 0;
  };

  const submitRegistration = async () => {
    try {
      const profilePicture = await getProfilePicture(
        formData.image,
        formData.firstName,
        formData.lastName
      );

      const response = await fetch(API_URL, {
        method: "POST",
        headers: { "Content-Type": "application/json" },
        body: JSON.stringify({
          firstName: formData.firstName,
          lastName: formData.lastName,
          phoneNumber: formData.phoneNumber,
          dateOfBirth: formData.dateOfBirth,
          gender: formData.gender,
          email: formData.email,
          password: formData.password,
          profilePicture,
        }),
      });

      const data = await response.json();

      if (response.ok) {
        navigate("/login");
      } else {
        const errorMessage =
          data.error || data.message || "Registration failed. Please try again.";
        setErrors({ submit: errorMessage });

        submit: "Server connection error. Ensure the backend is running.",
      });

      setStep((s) => s + 1);
    }
  };

    setErrors({});
    setStep((s) => s - 1);
  };

  const renderStep = () => {
    const stepProps = { formData, setFormData, errors };

    switch (step) {
      case 1:
        return <Step1 {...stepProps} />;
      case 2:
        return (
          <Step2
            {...stepProps}
            preview={preview}
            setPreview={setPreview}
            fileInputRef={fileInputRef}
          />
        );
      case 3:
        return <Step3 {...stepProps} />;
      case 4:
        return <Step4 {...stepProps} />;
      default:
        return null;
    }
  };

    <div className="copa-auth-container">
      <div className="copa-auth-card">

          <div className="d-inline-flex align-items-center gap-2">

            <span className="copa-brand-text">Copadrive</span>
          </div>

          <h1 className="h4 mt-3 mb-1">{currentConfig.title}</h1>
          <div className="text-secondary">{currentConfig.subtitle}</div>
        </div>

        {/* Form */}
        <form onSubmit={handleSubmit} className="d-grid gap-3">
          {renderStep()}

          {errors.submit && (
            <div className="alert alert-danger py-2 mb-0">{errors.submit}</div>
          )}

 * Handles all authentication-related API calls
 */

const API_BASE_URL = 'http://localhost:3000/api';

export const authenticateUser = async (email, password) => {
    try {
        const response = await fetch(`${API_BASE_URL}/tokens`, {
            method: 'POST',
            headers: { 'Content-Type': 'application/json' },
            body: JSON.stringify({ email, password })
        });

        const data = await response.json().catch(() => ({}));

        if (response.ok) {
            return { success: true, token: data.token, error: null };
        }

        if (response.status === 404) {
            return { success: false, error: "User not found. Please check your email or create an account." };
        }

        if (response.status === 401) {
            return { success: false, error: "Incorrect password. Please try again." };
        }

        return { success: false, error: data.error || "An error occurred on the server." };

    } catch (error) {
        console.error("Authentication error:", error);
        return { success: false, error: "Server connection error. Ensure the backend is running." };
    }
};

/**
 * Generates a default profile picture based on user's initials
 * @param {string} firstName - User's first name
 * @param {string} lastName - User's last name (optional)
 * @returns {string} - SVG data URL for default profile picture
 */
export const generateDefaultProfilePicture = (firstName, lastName = '') => {
    // Get initials (max 2 characters)
    const initials = (firstName.charAt(0) + (lastName ? lastName.charAt(0) : '')).toUpperCase();

    // Generate a consistent color based on the name
    const colors = [
        '#1a73e8', // Copa Blue
        '#34a853', // Copa Green
        '#ea4335', // Copa Red
        '#fbbc04', // Copa Yellow
        '#ff6d00', // Orange
        '#9334e6', // Purple
    ];

    const colorIndex = (firstName.charCodeAt(0) + (lastName ? lastName.charCodeAt(0) : 0)) % colors.length;
    const backgroundColor = colors[colorIndex];

    // Create SVG
    const svg = `
        <svg width="120" height="120" xmlns="http://www.w3.org/2000/svg">
            <circle cx="60" cy="60" r="60" fill="${backgroundColor}"/>
            <text x="50%" y="50%" dominant-baseline="middle" text-anchor="middle" 
                  font-family="Roboto, Arial, sans-serif" font-size="48" font-weight="400" fill="white">
                ${initials}
            </text>
        </svg>
    `;

    // Convert SVG to base64 data URL
    return `data:image/svg+xml;base64,${btoa(svg)}`;
};

/**
 * Gets the profile picture - either uploaded or default
 * @param {File|null} uploadedImage - The uploaded image file
 * @param {string} firstName - User's first name
 * @param {string} lastName - User's last name (optional)
 * @returns {Promise<string>} - Base64 string of the profile picture
 */
export const getProfilePicture = async (uploadedImage, firstName, lastName = '') => {
    if (uploadedImage) {
        // If user uploaded an image, convert it to base64
        return new Promise((resolve, reject) => {
            const reader = new FileReader();
            reader.readAsDataURL(uploadedImage);
            reader.onload = () => resolve(reader.result);
            reader.onerror = (error) => reject(error);
        });
    } else {
        // If no image uploaded, return default profile picture
        return generateDefaultProfilePicture(firstName, lastName);
    }
};

/**
 * Validates if the uploaded image is acceptable
 * @param {File} file - The image file
 * @returns {Object} - { valid: boolean, error: string|null }
 */
export const validateProfilePicture = (file) => {
    if (!file) {
        return { valid: true, error: null }; // No file is okay
    }

    // Check file size (max 5MB)
    if (file.size > 5000000) {
        return { valid: false, error: "Image size must be less than 5MB" };
    }

    // Check file type
    const validTypes = ['image/jpeg', 'image/jpg', 'image/png', 'image/gif', 'image/webp'];
    if (!validTypes.includes(file.type)) {
        return { valid: false, error: "Please upload a valid image file (JPEG, PNG, GIF, or WebP)" };
    }

    return { valid: true, error: null };
};
